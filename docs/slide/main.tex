% Fonts: Noto Sans (mac dinh cua theme) khong co trong MiKTeX nen deck se roi ve
% TeX Gyre Heros. Fira Sans hep hon Heros khoang 6% o dong tieu de nhung do dam
% net tuong duong, nen dat lai o day de tieu de dai khong bi tran dong.
%
% Hai luu y quan trong:
%  1. Khoi nay chi chay khi bien dich bang LuaLaTeX/XeLaTeX. Voi pdflatex thi
%     theme khong load fontspec, nen \setmainfont va \IfFontExistsTF khong ton
%     tai va se lam vo preamble. Vi vay phai kiem tra \ifdefined\setmainfont.
%     Voi pdflatex, theme tu dung package FiraSans, khong can dat gi o day.
%  2. Dung \IfFontExistsTF chu khong phai \IfFileExists, vi \IfFileExists tra
%     theo TEXINPUTS nen khong thay file .otf nam trong cay fonts cua MiKTeX.
\ifdefined\setmainfont
  \IfFontExistsTF{FiraSans-Regular.otf}{%
    \setmainfont{FiraSans-Regular.otf}[
      Ligatures=TeX,
      BoldFont=FiraSans-Bold.otf,
      ItalicFont=FiraSans-Italic.otf,
      BoldItalicFont=FiraSans-BoldItalic.otf
    ]%
    \setsansfont{FiraSans-Regular.otf}[
      Ligatures=TeX,
      BoldFont=FiraSans-Bold.otf,
      ItalicFont=FiraSans-Italic.otf,
      BoldItalicFont=FiraSans-BoldItalic.otf
    ]%
    \IfFontExistsTF{FiraMono-Regular.otf}{%
      \setmonofont{FiraMono-Regular.otf}[
        Scale=0.90,
        BoldFont=FiraMono-Bold.otf,
        ItalicFont=FiraMono-Oblique.otf,
        BoldItalicFont=FiraMono-BoldOblique.otf
      ]%
    }{}%
  }{}%
\fi
