\section{Điểm khác biệt}

\begin{frame}{Khoảng trống thị trường}{Chưa có trải nghiệm thuê thử trọn vẹn}
  \begin{itemize}\normalsize
    \item Chưa có \textbf{marketplace chuyên biệt} cho thuê gaming gear.
    \item Chưa hỗ trợ \textbf{thử tại nhà trước khi mua}.
    \item Chưa có \textbf{đặt lịch thuê} theo thiết bị.
    \item Thiếu \textbf{đánh giá sau nhiều ngày} trải nghiệm.
    \item Thiếu \textbf{giao nhận và bằng chứng} giảm tranh chấp.
    \item Chưa có \textbf{Cọc Trả Sau} giảm rào cản tài chính.
  \end{itemize}
\end{frame}
